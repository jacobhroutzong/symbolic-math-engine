cmake_minimum_required(VERSION 3.24)
project(SymbolicMathematics VERSION 0.1.0 LANGUAGES CXX)
add_library(ssm_core src/integer.cpp src/rational.cpp src/expression.cpp src/trace.cpp)
target_include_directories(ssm_core PUBLIC include)
target_compile_features(ssm_core PUBLIC cxx_std_20)
set_target_properties(ssm_core PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(ssm_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(ssm_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(symbolic_cli tools/cli.cpp)
target_link_libraries(symbolic_cli PRIVATE ssm_core)
include(CTest)
if(BUILD_TESTING)
  find_package(GTest CONFIG QUIET)
  if(NOT GTest_FOUND)
    set(SSM_GTEST_SOURCE "" CACHE PATH "Existing GoogleTest source, no auto-download")
    if(NOT EXISTS "${SSM_GTEST_SOURCE}/CMakeLists.txt")
      message(FATAL_ERROR "Supply GTest or SSM_GTEST_SOURCE, or BUILD_TESTING=OFF")
    endif()
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    add_subdirectory("${SSM_GTEST_SOURCE}" "${CMAKE_BINARY_DIR}/googletest" EXCLUDE_FROM_ALL)
  endif()
  add_executable(symbolic_tests tests/core_tests.cpp tests/trace_tests.cpp)
  target_link_libraries(symbolic_tests PRIVATE ssm_core GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(symbolic_tests)
  set(SSM_REFERENCE_PYTHON "" CACHE FILEPATH "Optional existing Python with SymPy reference")
  if(SSM_REFERENCE_PYTHON)
    add_test(NAME Symbolic.IndependentReference COMMAND "${SSM_REFERENCE_PYTHON}" -B
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/reference.py" --cli $<TARGET_FILE:symbolic_cli>
      --reference-path "${CMAKE_CURRENT_SOURCE_DIR}/.reference-python")
  endif()
endif()
